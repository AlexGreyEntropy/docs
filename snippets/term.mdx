export const Term = ({ word, part, phonetic, example, id, children }) => {
  const escaped = word.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");
  const pieces = example ? example.split(new RegExp(`(${escaped})`, "i")) : [];
  return (
    <div
      id={id}
      style={{
        borderLeft: "2px solid currentColor",
        padding: "0.25rem 0 0.25rem 1.25rem",
        margin: "1.75rem 0",
      }}
    >
      <div style={{ display: "flex", alignItems: "baseline", gap: "0.6rem", flexWrap: "wrap" }}>
        <span style={{ fontSize: "1.75rem", fontWeight: 800, letterSpacing: "-0.04em", lineHeight: 1.1 }}>
          {word}
        </span>
        {part && <span style={{ opacity: 0.5, fontSize: "0.95rem" }}>({part})</span>}
      </div>
      {phonetic && (
        <div style={{ fontFamily: "Georgia, serif", fontStyle: "italic", opacity: 0.5, marginTop: "0.25rem" }}>
          {phonetic}
        </div>
      )}
      <div style={{ fontFamily: "Georgia, serif", fontSize: "1.05rem", lineHeight: 1.6, marginTop: "0.75rem" }}>
        {children}
      </div>
      {example && (
        <div style={{ fontFamily: "Georgia, serif", marginTop: "0.6rem", opacity: 0.75 }}>
          <span style={{ opacity: 0.8 }}>e.g.: </span>
          {pieces.map((p, i) =>
            p.toLowerCase() === word.toLowerCase() ? (
              <span key={i} style={{ textDecoration: "underline", textUnderlineOffset: "3px", fontWeight: 600 }}>
                {p}
              </span>
            ) : (
              <span key={i}>{p}</span>
            )
          )}
        </div>
      )}
    </div>
  );
};
